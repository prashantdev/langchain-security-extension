161    def evaluate(self, request: PolicyEvaluationRequest) -> PolicyDecision:
162        """Synchronously evaluate policy request against remote OPA endpoint.
163
164        Args:
165            request: Evaluation request envelope.
166
167        Returns:
168            PolicyDecision: Remote PDP decision verdict.
169        """
170        current_state = self.state
171        if current_state == CircuitBreakerState.OPEN:
172            return self._handle_fail_closed(
173                request=request,
174                reason="Remote PDP circuit breaker OPEN (fail-closed) [CIRCUIT_BREAKER_OPEN | LOCAL_PDP_FALLBACK]",
175            )
176
177        payload = self._build_payload(request)
178
179        try:
180            import urllib.request
181            import json
182
183            data = json.dumps(payload).encode("utf-8")
184            req = urllib.request.Request(
185                self._endpoint_url,
186                data=data,
187                headers={"Content-Type": "application/json"},
188                method="POST",
189            )
190            with urllib.request.urlopen(req, timeout=self._timeout_seconds) as resp:
191                if resp.status == 200:
192                    resp_data = json.loads(resp.read().decode("utf-8"))
193                    result = resp_data.get("result", {})
194                    allow = result.get("allow", False) if isinstance(result, dict) else bool(result)
195                    self._record_success(request)
196                    if allow:
197                        return PolicyDecision.permit("Authorized by Remote OPA PDP", {})
198                    return PolicyDecision.deny("Denied by Remote OPA PDP", {})
199
200            self._record_failure(request, "HTTP non-200 response")
201            return self._handle_fail_closed(
202                request=request,
203                reason="Remote OPA PDP returned invalid HTTP response",
204            )
205
206        except Exception as err:
207            self._record_failure(request, str(err))
208            return self._handle_fail_closed(
209                request=request,
210                reason=f"Remote OPA PDP request error: {str(err)}",
211            )

261    def _build_payload(self, request: PolicyEvaluationRequest) -> Dict[str, Any]:
262        safe_context = {k: v for k, v in request.context.items() if k != "config"}
263        return {
264            "input": {
265                "subject": request.subject.to_dict(),
266                "action": request.action,
267                "resource": request.resource,
268                "context": safe_context,
269            }
270        }

272    def _record_success(self, request: PolicyEvaluationRequest) -> None:
273        prev_state = self._state
274        self._consecutive_failures = 0
275        self._state = CircuitBreakerState.CLOSED
276        self._opened_at = None
277        if prev_state != CircuitBreakerState.CLOSED:
278            self._emit_circuit_event(
279                request=request,
280                reason_code=EVENT_CIRCUIT_BREAKER_TRANSITION,
281                diagnostic=f"{EVENT_CIRCUIT_BREAKER_TRANSITION}: {prev_state.value} -> {CircuitBreakerState.CLOSED.value}",
282            )
283
284    def _record_failure(self, request: PolicyEvaluationRequest, error_detail: str) -> None:
285        prev_state = self._state
286        self._consecutive_failures += 1
287        if prev_state == CircuitBreakerState.HALF_OPEN or self._consecutive_failures >= self._failure_threshold:
288            self._state = CircuitBreakerState.OPEN
289            self._opened_at = time.time()
290            self._emit_circuit_event(
291                request=request,
292                reason_code=EVENT_CIRCUIT_BREAKER_OPEN,
293                diagnostic=(
294                    f"{EVENT_CIRCUIT_BREAKER_OPEN} | {EVENT_CIRCUIT_BREAKER_TRANSITION} "
295                    f"({prev_state.value} -> {self._state.value}) | LOCAL_PDP_FALLBACK: {error_detail}"
296                ),
297            )
298
299    def _emit_circuit_event(self, request: PolicyEvaluationRequest, reason_code: str, diagnostic: str) -> None:
300        req_config = request.context.get("config") if isinstance(request.context, dict) else None
301        if isinstance(req_config, dict) and isinstance(req_config.get("configurable"), dict):
302            req_config["configurable"]["sec:circuit_breaker_state"] = self._state.value
303            req_config["configurable"]["sec:circuit_breaker_transition"] = diagnostic
304
305        mgr = self.audit_manager
306        if mgr:
307            event = AuditEvent.create(
308                subject_id=request.subject.subject_id,
309                tenant_id=request.subject.tenant_id,
310                action=request.action,
311                resource=str(request.resource.get("name") or request.resource.get("target_id") or "RemotePDPAdapter"),
312                decision=DECISION_DENY,
313                diagnostic_reason=diagnostic,
314                severity=SEVERITY_WARNING,
315                metadata={"circuit_state": self._state.value},
316                reason_code=reason_code,
317            )
318            mgr.record_event(event)
319
320    def _handle_fail_closed(self, request: PolicyEvaluationRequest, reason: str) -> PolicyDecision:
321        req_config = request.context.get("config") if isinstance(request.context, dict) else None
322        if isinstance(req_config, dict) and isinstance(req_config.get("configurable"), dict):
323            req_config["configurable"]["sec:circuit_breaker_state"] = self._state.value
324
325        diag_reason = reason
326        if self._state == CircuitBreakerState.OPEN and EVENT_CIRCUIT_BREAKER_OPEN not in diag_reason:
327            diag_reason = f"{reason} [{EVENT_CIRCUIT_BREAKER_OPEN} | {EVENT_CIRCUIT_BREAKER_TRANSITION} | LOCAL_PDP_FALLBACK]"
328
329        return PolicyDecision.deny(
330            diagnostic_reason=diag_reason,
331            obligations={},
332        )
